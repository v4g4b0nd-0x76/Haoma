At 1 the map cannot zoom out past fit, which keeps regional maps from letterboxing; pass a value below 1 only when the overview should pull farther back. Pan is off until the view is zoomed in. eras — optional comma list ordering era layers, e.g. eras="lands,achaemenid,parthian,myth" (discovered from body when omitted). No external JS/CSS dependencies. */}} {{- /* Every param is coerced to a string with printf: .Get returns interface{}, and calling string methods (TrimSpace &c.) on it is what broke older builds of this shortcode on Hugo 0.123. default first, coerce after, so nil params still fall back. */ -}} {{- $id := .Get "id" | default (printf "map-%d" now.Unix) | printf "%v" -}} {{- $src := .Get "src" | default "" | printf "%v" -}} {{- $w := .Get "width" | default "100%" | printf "%v" -}} {{- $h := .Get "height" | default "480px" | printf "%v" -}} {{- $theme := .Get "theme" | default "dark" | printf "%v" -}} {{- $caption := .Get "caption" | default "" | printf "%v" -}} {{- $route := .Get "route" | default "" | printf "%v" -}} {{- $pulse := .Get "pulse" | default "true" | printf "%v" -}} {{- $minzoom := .Get "minzoom" | default "1" | printf "%v" -}} {{- $erasIn := .Get "eras" | default "" | printf "%v" -}} {{- /* Regional maps stay at fit: zooming out only letterboxes.
