{{- /* Print an 11-char YouTube id from a bare id or common watch/embed/shorts URL. */ -}} {{- $raw := trim (. | default "") " \t\n\r" -}} {{- $id := "" -}} {{- if findRE `^[A-Za-z0-9_-]{11}$` $raw -}} {{- $id = $raw -}} {{- else if findRE `(?i)(?:youtube\.com|youtu\.be)` $raw -}} {{- $id = replaceRE `(?i)^.*(?:youtu\.be/|youtube\.com/(?:embed/|v/|shorts/|live/|watch\?(?:[^#]*&)?v=))([A-Za-z0-9_-]{11}).*$` `$1` $raw -}} {{- if not (findRE `^[A-Za-z0-9_-]{11}$` $id) }}{{ $id = "" }}{{ end -}} {{- end -}} {{- $id -}}
